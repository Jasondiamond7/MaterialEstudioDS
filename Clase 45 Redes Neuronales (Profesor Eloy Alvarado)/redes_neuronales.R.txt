# 02. Seguir un gradiente — diapositivas 19–24
theta <- c(.4, .1, .8, -.2)   # w1, b1, w2, b2.
tasa_ejemplo <- .1
perdida_pequena <- function(t) {
  s <- t[3] * max(t[1] * .5 + t[2], 0) + t[4]
  log1p(exp(-s))
}
a1 <- theta[1] * .5 + theta[2]
h <- max(a1, 0)
delta_salida <- plogis(theta[3] * h + theta[4]) - 1
delta_oculta <- delta_salida * theta[3] * (a1 > 0)
gradiente <- c(delta_oculta * .5, delta_oculta, delta_salida * h, delta_salida)
numerico <- vapply(seq_along(theta), function(i) {
  paso <- diag(length(theta))[, i] * 1e-6
  (perdida_pequena(theta + paso) - perdida_pequena(theta - paso)) / 2e-6
}, numeric(1))
stopifnot(max(abs(gradiente - numerico)) < 1e-8)
actualizados <- theta - tasa_ejemplo * gradiente
cat("Gradiente:", gradiente, "\nPérdida antes/después:",
    perdida_pequena(theta), perdida_pequena(actualizados), "\n")

# Funciones de entrenamiento. Solo usan R base.
# Cada fila es una imagen; W[[j]] conecta la capa j con la siguiente.
activar <- function(z, tipo) if (tipo == "relu") pmax(z, 0) else tanh(z)
softmax <- function(z) {
  e <- exp(z - apply(z, 1, max))
  e / rowSums(e)
}
propagar <- function(x, modelo) {
  a <- list(x)
  for (j in seq_along(modelo$W)) {
    z <- sweep(a[[j]] %*% modelo$W[[j]], 2, modelo$b[[j]], "+")
    a[[j + 1L]] <- if (j < length(modelo$W)) activar(z, modelo$activacion) else
      if (modelo$salida == "softmax") softmax(z) else z
  }
  a
}
predecir <- function(x, modelo) tail(propagar(x, modelo), 1)[[1]]
crear_red <- function(anchos, activacion, salida, semilla) {
  set.seed(semilla)
  W <- b <- list()
  for (j in seq_len(length(anchos) - 1L)) {
    limite <- sqrt(6 / (anchos[j] + anchos[j + 1L]))
    W[[j]] <- matrix(runif(anchos[j] * anchos[j + 1L], -limite, limite),
                     anchos[j], anchos[j + 1L])
    b[[j]] <- runif(anchos[j + 1L], -limite, limite)
  }
  list(W = W, b = b, activacion = activacion, salida = salida)
}
gradientes <- function(x, objetivo, modelo, alpha) {
  a <- propagar(x, modelo)
  L <- length(modelo$W)
  # Softmax + entropía cruzada, o salida lineal + mitad de suma de cuadrados.
  # Para reconstrucción, el MSE por píxel se calcula aparte al validar.
  delta <- (a[[L + 1L]] - objetivo) / nrow(x)
  gW <- gb <- vector("list", L)
  for (j in L:1L) {
    gW[[j]] <- crossprod(a[[j]], delta) + alpha * modelo$W[[j]] / nrow(x)
    gb[[j]] <- colSums(delta)
    if (j > 1L) {
      derivada <- if (modelo$activacion == "relu") (a[[j]] > 0) else 1 - a[[j]]^2
      delta <- (delta %*% t(modelo$W[[j]])) * derivada
    }
  }
  list(W = gW, b = gb)
}
entrenar <- function(ocultas, salida, objetivo, nombre) {
  modelo <- crear_red(c(ncol(X), ocultas, ncol(objetivo)), ACTIVACION, salida, SEMILLA)
  # Adam conserva dos promedios del gradiente a lo largo de todos los lotes.
  mW <- vW <- lapply(modelo$W, function(w) w * 0)
  mb <- vb <- lapply(modelo$b, function(b) b * 0)
  paso <- 0L
  mejor <- Inf
  historial <- data.frame(epoca = seq_len(EPOCAS), entrenamiento = NA_real_,
                          validacion = NA_real_)
  evaluar <- function(ind, red) {
    p <- predecir(X[ind, , drop = FALSE], red)
    if (salida == "softmax") -mean(rowSums(objetivo[ind, , drop = FALSE] *
                                            log(pmax(p, 1e-15)))) else
      mean((objetivo[ind, , drop = FALSE] - p)^2)
  }
  inicial <- evaluar(which(tr), modelo)
  for (epoca in seq_len(EPOCAS)) {
    orden <- sample(which(tr))
    for (inicio in seq.int(1L, length(orden), by = LOTE)) {
      ind <- orden[inicio:min(inicio + LOTE - 1L, length(orden))]
      g <- gradientes(X[ind, , drop = FALSE], objetivo[ind, , drop = FALSE], modelo, ALPHA)
      paso <- paso + 1L
      tasa_paso <- TASA * sqrt(1 - .999^paso) / (1 - .9^paso)
      for (j in seq_along(modelo$W)) {
        mW[[j]] <- .9 * mW[[j]] + .1 * g$W[[j]]
        vW[[j]] <- .999 * vW[[j]] + .001 * g$W[[j]]^2
        mb[[j]] <- .9 * mb[[j]] + .1 * g$b[[j]]
        vb[[j]] <- .999 * vb[[j]] + .001 * g$b[[j]]^2
        modelo$W[[j]] <- modelo$W[[j]] - tasa_paso * mW[[j]] / (sqrt(vW[[j]]) + 1e-8)
        modelo$b[[j]] <- modelo$b[[j]] - tasa_paso * mb[[j]] / (sqrt(vb[[j]]) + 1e-8)
      }
    }
    err_tr <- evaluar(which(tr), modelo)
    err_va <- evaluar(which(va), modelo)
    if (!all(is.finite(c(err_tr, err_va)))) stop("Ajuste no finito: revisa la tasa y los datos.")
    historial[epoca, 2:3] <- c(err_tr, err_va)
    if (err_va < mejor) {
      mejor <- err_va
      elegido <- modelo           # Copia de los parámetros de esta época.
      epoca_elegida <- epoca
    }
    if (epoca == 1L || epoca %% 20L == 0L || epoca == EPOCAS)
      cat(nombre, "| época", epoca, "| entrenamiento", signif(err_tr, 6),
          "| validación", signif(err_va, 6), "\n")
  }
  list(modelo = elegido, historial = historial, epoca = epoca_elegida,
       validacion = mejor, inicial = inicial,
       parametros = sum(lengths(elegido$W)) + sum(lengths(elegido$b)))
}
metricas <- function(reales, p) {
  predicha <- max.col(p, ties.method = "first") - 1L
  data.frame(log_loss = -mean(log(pmax(p[cbind(seq_along(reales), reales + 1L)], 1e-15))),
             exactitud = mean(predicha == reales), errores = sum(predicha != reales))
}
guardar_tabla <- function(tabla, nombre)
  write.csv(tabla, file.path(OUT, nombre), row.names = FALSE, fileEncoding = "UTF-8")
figura <- function(nombre, ancho = 1200, alto = 600)
  png(file.path(OUT, nombre), width = ancho, height = alto, res = 140)
curvas <- function(ajustes, nombres) {
  figura("curvas.png", 650 * length(ajustes), 550)
  par(mfrow = c(1, length(ajustes)), mar = c(4, 4, 3, 1))
  for (i in seq_along(ajustes)) {
    h <- ajustes[[i]]$historial
    matplot(h$epoca, h[, 2:3], type = "l", lty = 1, col = c("#126782", "#d17a22"),
            xlab = "Época", ylab = "Pérdida sin penalización", main = nombres[i])

    legend("topright", c("Entrenamiento", "Validación"), lty = 1,
           col = c("#126782", "#d17a22"), bty = "n")
  }
  invisible(dev.off())
}
# Regresión logística multiclase de referencia: C=1.
# Criterio = suma de log loss + 1/2 suma(W^2); los interceptos no se penalizan.
# La estandarización, si se solicita, se estima exclusivamente con entrenamiento.
ajustar_logistica <- function(x, etiquetas, estandarizar = FALSE) {
  centro <- if (estandarizar) colMeans(x) else rep(0, ncol(x))
  escala <- if (estandarizar) sqrt(colMeans(sweep(x, 2, centro)^2)) else rep(1, ncol(x))
  escala[escala == 0] <- 1
  z <- cbind(sweep(sweep(x, 2, centro), 2, escala, "/"), 1)
  objetivo <- diag(10)[etiquetas + 1L, , drop = FALSE]
  evaluar <- function(v, gradiente = FALSE) {
    w <- matrix(v, ncol = 10)
    puntajes <- z %*% w
    maximo <- apply(puntajes, 1, max)
    log_suma <- maximo + log(rowSums(exp(puntajes - maximo)))
    penalizados <- w
    penalizados[nrow(w), ] <- 0
    if (gradiente) return(as.vector(crossprod(z, softmax(puntajes) - objetivo) + penalizados))
    sum(log_suma - rowSums(objetivo * puntajes)) + sum(penalizados^2) / 2
  }
  ajuste <- optim(rep(0, ncol(z) * 10), evaluar, function(v) evaluar(v, TRUE),
                   method = "BFGS", control = list(maxit = 1500, reltol = 1e-10))
  if (ajuste$convergence != 0) warning("La logística alcanzó su límite de optimización.")
  list(W = matrix(ajuste$par, ncol = 10), centro = centro, escala = escala)
}
predecir_logistica <- function(x, ajuste) {
  z <- cbind(sweep(sweep(x, 2, ajuste$centro), 2, ajuste$escala, "/"), 1)
  softmax(z %*% ajuste$W)
}
