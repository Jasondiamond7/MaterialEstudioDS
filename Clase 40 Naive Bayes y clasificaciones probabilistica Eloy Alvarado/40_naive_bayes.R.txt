buscar_datos <- function() {
  archivos <- unlist(lapply(sys.frames(), function(fr) fr$ofile), use.names=FALSE)
  archivos <- c(rev(archivos), sub("^--file=", "", grep("^--file=",commandArgs(),value=TRUE)))
  inicios <- if(nzchar(RUTA_MATERIAL)) RUTA_MATERIAL else c(archivos,getwd())
  for(inicio in inicios) {
    if(!file.exists(inicio)) next
    base <- normalizePath(inicio,winslash="/",mustWork=TRUE)
    if(!dir.exists(base)) base <- dirname(base)
    padres <- base
    while(!nzchar(RUTA_MATERIAL) && dirname(base)!=base) {
      base <- dirname(base); padres <- c(padres,base)
    }
    for(padre in padres) for(carpeta in c(padre,file.path(padre,"datos"),
      file.path(padre,"clase/datos"),file.path(padre,"sesiones/40_2026-10-01_naive_bayes/clase/datos"))) {
      if(file.exists(file.path(carpeta,"spam_s40.csv"))) return(carpeta)
    }
  }
  stop("Guardar spam_s40.csv junto al código o completar RUTA_MATERIAL.")
}

posterior <- function(log_conjunta) {
  z <- sweep(log_conjunta,1,apply(log_conjunta,1,max),"-")
  p <- exp(z); p/rowSums(p)
}
log_posterior <- function(log_conjunta) {
  z <- sweep(log_conjunta,1,apply(log_conjunta,1,max),"-")
  sweep(z,1,log(rowSums(exp(z))),"-")
}
medir <- function(y,p,t=.5,logp=NULL) {
  q <- pmin(pmax(p,1e-15),1-1e-15); pred <- p>=t
  if(is.null(logp)) logp <- cbind(log1p(-q),log(q))
  perdida <- -mean(pmax(logp[cbind(seq_along(y),y+1)],log(1e-15)))
  vn <- sum(y==0 & !pred); fp <- sum(y==0 & pred)
  fn <- sum(y==1 & !pred); vp <- sum(y==1 & pred)
  n1 <- sum(y==1); n0 <- sum(y==0)
  auc <- (sum(rank(logp[,2]-logp[,1],ties.method="average")[y==1])-n1*(n1+1)/2)/(n1*n0)
  data.frame(log_loss=perdida,brier=mean((y-p)^2),
    auc=auc,accuracy=mean(pred==y),VN=vn,FP=fp,FN=fn,VP=vp,costo_1000=1000*(5*fp+fn)/length(y))
}

# La tabla siguiente automatiza el mismo cálculo, con y sin suavizado.
# theta: probabilidad de presencia; primera fila legítimo, segunda spam.
n <- c(80,20)
conteos <- rbind(c(8,16),c(16,14)) # oferta, enlace
x_nuevo <- c(1,0)
filas <- lapply(c(0,1),function(alfa) {
  theta <- (conteos+alfa)/(n+2*alfa)
  logc <- log(n/sum(n))+rowSums(sweep(log(theta),2,x_nuevo,"*")+sweep(log1p(-theta),2,1-x_nuevo,"*"))
  data.frame(alfa=alfa,theta_oferta_0=theta[1,1],theta_enlace_0=theta[1,2],
    theta_oferta_1=theta[2,1],theta_enlace_1=theta[2,2],posterior=posterior(matrix(logc,nrow=1))[1,2])
})
tabla("02_bernoulli",do.call(rbind,filas))

# %% 05. Spambase y protocolo ----
datos <- read.csv(file.path(DATOS,"spam_s40.csv"),check.names=FALSE)
variables <- grep("^(word_freq_|char_freq_)",names(datos),value=TRUE)
stopifnot(length(variables)==54,!anyNA(datos))
ent <- datos[datos$conjunto=="entrenamiento",]; prueba <- datos[datos$conjunto=="prueba",]

  tipo=c("prior","bern","bern","bern","gauss"),valor=c(0,.5,1,2,1e-9))
NOMBRES <- c(FRECUENCIA="Frecuencia de spam",BERN_05="Bernoulli: alfa 0.5",BERN_1="Bernoulli: alfa 1",BERN_2="Bernoulli: alfa 2",GAUSS_LOG="Gaussiana: log(1+x)")

ajustar_predecir <- function(xa,ya,xv,tipo,valor) {
  if(tipo=="prior") {
    p <- rep(mean(ya),nrow(xv))
    return(list(p=p,logp=cbind(log1p(-p),log(p))))
  }
  prior <- c(mean(ya==0),mean(ya==1))
  if(tipo=="bern") {
    a <- 1*(xa>0); v <- 1*(xv>0)
    theta <- t(vapply(0:1,function(k) (colSums(a[ya==k,,drop=FALSE])+valor)/(sum(ya==k)+2*valor),numeric(ncol(xa))))
    logc <- v %*% t(log(theta))+(1-v) %*% t(log1p(-theta))
    logc <- sweep(logc,2,log(prior),"+")
    lp <- log_posterior(logc)
    return(list(p=exp(lp[,2]),logp=lp,theta=theta,prior=prior))
  }
  a <- log1p(xa); v <- log1p(xv)
  # Varianza de máxima verosimilitud: divisor n, como GaussianNB.
  var_ml <- function(m) colMeans(sweep(m,2,colMeans(m),"-")^2)
  epsilon <- valor*max(var_ml(a))
  mu <- t(vapply(0:1,function(k) colMeans(a[ya==k,,drop=FALSE]),numeric(ncol(a))))
  varianza <- t(vapply(0:1,function(k) var_ml(a[ya==k,,drop=FALSE])+epsilon,numeric(ncol(a))))
  logc <- sapply(1:2,function(k) {
    d <- sweep(v,2,mu[k,],"-")
    log(prior[k])-.5*sum(log(2*pi*varianza[k,]))-.5*rowSums(sweep(d^2,2,varianza[k,],"/"))
  })
  lp <- log_posterior(logc)
  list(p=exp(lp[,2]),logp=lp,mu=mu,varianza=varianza,prior=prior)
}
particion <- aggregate(id~conjunto+spam,datos,length); names(particion)[3] <- "n"
particion <- particion[order(particion$conjunto,particion$spam),]
tabla("05_particion",particion)

# %% 06. Predicciones fuera de pliegue y comparación ----
oof <- list(); logs_oof <- list(); comparacion <- list()
for(j in seq_len(nrow(RECETAS))) {
  r <- RECETAS[j,]; pred <- numeric(length(y)); lp <- matrix(0,length(y),2)
  for(k in sort(unique(folds))) {
    val <- folds==k
    ajuste <- ajustar_predecir(X[!val,,drop=FALSE],y[!val],X[val,,drop=FALSE],r$tipo,r$valor)
    pred[val] <- ajuste$p; lp[val,] <- ajuste$logp
  }
  oof[[r$modelo]] <- pred
  logs_oof[[r$modelo]] <- lp
  comparacion[[j]] <- cbind(modelo=r$modelo,medir(y,pred,logp=lp))
}
comparacion <- do.call(rbind,comparacion)
tabla("06_comparacion_oof",comparacion)
tabla("06_predicciones_oof",data.frame(id=ent$id,y=y,fold=folds,oof,check.names=FALSE))
ganador <- comparacion$modelo[which.min(comparacion$log_loss)]
cat("Selección por log loss OOF:",NOMBRES[ganador],"\n")
figura("06_comparacion",function() {
  par(mar=c(4,10,1,1))
  barplot(rev(comparacion$log_loss),names.arg=rev(NOMBRES[comparacion$modelo]),horiz=TRUE,las=1,
    col=rev(ifelse(comparacion$modelo==ganador,VERDE,AZUL)),xlab="Log loss OOF (menor es mejor)")
})

# %% 08. Probabilidades por clase para las señales de un correo ----
modelo_bern <- ajustar_predecir(X,y,X,"bern",1)
indice <- which.min(ent$id); b <- 1*(X[indice,]>0); th <- modelo_bern$theta
aportes <- b*log(th[2,]/th[1,])+(1-b)*log((1-th[2,])/(1-th[1,]))
prior_log <- log(mean(y)/(1-mean(y)))
# Contar el estado observado: presencia si b=1, ausencia si b=0.
# Alfa=1 agrega una unidad al conteo y dos al denominador por clase.
coincide <- sweep(1*(X>0),2,b,"==")
n0 <- sum(y==0); n1 <- sum(y==1)
cuenta0 <- colSums(coincide[y==0,,drop=FALSE])
cuenta1 <- colSums(coincide[y==1,,drop=FALSE])
prob0 <- (cuenta0+1)/(n0+2); prob1 <- (cuenta1+1)/(n1+2)
explicacion <- data.frame(variable=variables,presente=b,conteo_legitimo=cuenta0,
  n_legitimo=n0,conteo_spam=cuenta1,n_spam=n1,prob_legitimo=prob0,
  prob_spam=prob1,aporte_log_odds=aportes,row.names=NULL)
tabla("08_aportes",explicacion)
tabla("08_correo",data.frame(id=ent$id[indice],log_odds_prior=prior_log,suma_aportes=sum(aportes),posterior=modelo_bern$p[indice]))
top <- explicacion[order(-abs(aportes),method="radix")[1:8],]; top <- top[order(top$aporte_log_odds),]
# El detalle logarítmico permite reconstruir la probabilidad completa.
# Para leer el gráfico basta comparar las probabilidades dentro de cada clase.

  barplot(rbind(top$prob_legitimo,top$prob_spam),names.arg=etiquetas,horiz=TRUE,
