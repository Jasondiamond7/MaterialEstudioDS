carpeta_clase <- function() {
    buscar <- function(inicio, subir = TRUE) {
        if (!nzchar(inicio) || !file.exists(path.expand(inicio))) return(NULL)
        base <- normalizePath(path.expand(inicio), winslash = "/", mustWork = TRUE)
        if (!dir.exists(base)) base <- dirname(base)
        padres <- base
        while (subir && dirname(base) != base) {
            base <- dirname(base); padres <- c(padres, base)
        }
        for (padre in padres) for (candidata in c(padre, file.path(padre, "clase"),
            file.path(padre, "sesiones/38_2026-09-24_evaluacion_seleccion/clase"))) {
            for (datos in c(candidata, file.path(candidata, "datos"))) {
                archivos <- file.path(datos, c("hormigon_s38.csv", "spam_s38.csv"))
                presentes <- file.exists(archivos) & !dir.exists(archivos)
                if (all(presentes)) return(list(
                    material = normalizePath(candidata, winslash = "/", mustWork = TRUE),
                    datos = normalizePath(datos, winslash = "/", mustWork = TRUE)))
                if (any(presentes)) stop("Material incompleto en ", datos, ": faltan ",
                    paste(basename(archivos[!presentes]), collapse = ", "),
                    ". Guardar ambos CSV juntos.", call. = FALSE)
            }
            if (file.exists(file.path(candidata, "DipDS_2026_S38.pdf")))
                stop("Material incompleto en ", candidata,
                    ": faltan hormigon_s38.csv y spam_s38.csv.", call. = FALSE)
        }
        NULL
    }
    manual <- trimws(RUTA_MATERIAL)
    if (nzchar(manual)) {
        encontrada <- buscar(manual, subir = FALSE)
        if (!is.null(encontrada)) return(encontrada)
        stop("RUTA_MATERIAL no contiene el material completo: ", manual,
            "\nIndicar la carpeta con hormigon_s38.csv y spam_s38.csv.",
            call. = FALSE)
    }
    archivos <- unlist(lapply(sys.frames(), function(fr) fr$ofile), use.names = FALSE)
    archivos <- c(rev(archivos), sub("^--file=", "", grep("^--file=", commandArgs(), value = TRUE)))
    for (inicio in c(archivos, getwd(), Sys.getenv("DIPDS_WORKSPACE"))) {
        encontrada <- buscar(inicio)
        if (!is.null(encontrada)) return(encontrada)
    }
    stop("No se encontró el material desde ", getwd(),
        "\nTrabajar desde la carpeta de los scripts y CSV, o completar RUTA_MATERIAL en el tramo 00.",
        "\nConservar hormigon_s38.csv y spam_s38.csv junto a los scripts.",
        call. = FALSE)
}

# Claves internas de los CSV; los nombres legibles se usan al mostrar resultados.
nombre_modelo <- function(clave) {
    nombres <- c(MEDIA="Media de entrenamiento", LINEAL="Regresión lineal",
        FRECUENCIA="Frecuencia de spam", LOGISTICA_C1="Logística (C=1)",
        BOSQUE_8="Bosque (profundidad 8)", SIEMPRE_NO_SPAM="Siempre no spam")
    vapply(clave, function(z) if (startsWith(z,"Q_A"))
        paste0("Cuadrática: alfa=",sub("Q_A","",z)) else unname(nombres[z]), character(1))
}
mostrar <- function(tabla) {
    t <- tabla
    if ("modelo" %in% names(t)) t$modelo <- nombre_modelo(t$modelo)
    print(t, row.names = FALSE, digits = 4)
}

metricas <- function(real, predicha) {
    e <- real - predicha
    list(RMSE = sqrt(mean(e^2)), MAE = mean(abs(e)),
         R2 = 1 - sum(e^2) / sum((real - mean(real))^2))
}
escala_ajuste <- function(x) {
    x <- as.matrix(x); centro <- colMeans(x)
    z <- sweep(x, 2, centro, "-")
    escala <- sqrt(colMeans(z^2)); escala[escala == 0] <- 1
    list(centro = centro, escala = escala)
}
transformar <- function(x, e) sweep(sweep(as.matrix(x), 2, e$centro, "-"), 2, e$escala, "/")
cuadratico <- function(x) {
    # Mismo orden que PolynomialFeatures(2): 8 lineales y 36 productos i <= j.
    pares <- list()
    for (i in seq_len(ncol(x))) for (j in i:ncol(x))
        pares[[length(pares) + 1L]] <- x[, i] * x[, j]
    cbind(x, do.call(cbind, pares))
}
ajustar_regresion <- function(nombre, x, y) {
    obj <- list(nombre = nombre, media = mean(y))
    if (nombre == "MEDIA") return(obj)
    obj$e1 <- escala_ajuste(x); z <- transformar(x, obj$e1)
    if (startsWith(nombre, "Q_A")) {
        z <- cuadratico(z); obj$e2 <- escala_ajuste(z); z <- transformar(z, obj$e2)
    }
    obj$centro <- colMeans(z); z <- sweep(z, 2, obj$centro, "-")
    des <- svd(z); alfa <- if (nombre == "LINEAL") 0 else as.numeric(sub("Q_A", "", nombre))
    peso <- if (alfa > 0) des$d / (des$d^2 + alfa) else
        ifelse(des$d > max(dim(z)) * .Machine$double.eps * max(des$d), 1/des$d, 0)
    obj$beta <- drop(des$v %*% (peso * drop(crossprod(des$u, y - obj$media))))
    obj
}
predecir_regresion <- function(obj, x) {
    if (obj$nombre == "MEDIA") return(rep(obj$media, nrow(x)))
    z <- transformar(x, obj$e1)
    if (!is.null(obj$e2)) z <- transformar(cuadratico(z), obj$e2)
    drop(obj$media + sweep(z, 2, obj$centro, "-") %*% obj$beta)
}
curvas <- function(y, p) {
    orden <- order(-p); yy <- y[orden]; pp <- p[orden]
    # Procesar juntos los empates de probabilidad.
    finales <- which(!duplicated(pp, fromLast = TRUE))
    tp <- cumsum(yy)[finales]; fp <- finales - tp
    list(recall = c(0, tp/sum(y)), fpr = c(0, fp/sum(y == 0)),
         precision = c(1, tp/finales))
}
probabilisticas <- function(y, p) {
    q <- pmin(pmax(p, .Machine$double.eps), 1 - .Machine$double.eps)
    curva <- curvas(y, p); n1 <- sum(y == 1); n0 <- sum(y == 0)
    list(log_loss = -mean(y*log(q) + (1-y)*log1p(-q)), Brier = mean((y-p)^2),
         ROC_AUC = (sum(rank(p)[y == 1]) - n1*(n1+1)/2)/(n1*n0),
         AP = sum(diff(curva$recall) * curva$precision[-1]))
}
decision <- function(y, p, t) {
    pred <- as.integer(p >= t)
    vn <- sum(y == 0 & pred == 0); fp <- sum(y == 0 & pred == 1)
    fn <- sum(y == 1 & pred == 0); vp <- sum(y == 1 & pred == 1)
    list(umbral = t, VN = vn, FP = fp, FN = fn, VP = vp,
         accuracy = (vn+vp)/length(y), precision = if (vp+fp) vp/(vp+fp) else 0,
         recall = vp/(vp+fn), F1 = 2*vp/(2*vp+fp+fn), costo = (5*fp+fn)/length(y))
}
ajustar_clasificador <- function(nombre, x, y) {
    if (nombre == "FRECUENCIA") return(list(nombre = nombre, media = mean(y)))
    if (nombre == "LOGISTICA_C1") {
        e <- escala_ajuste(x); z <- cbind(1, transformar(x, e))
        # Misma pérdida logística L2: suma de log loss + ||beta||^2 / (2*C), C=1.
        # El intercepto no se penaliza; se divide todo por n para optimizar.
        fn <- function(b) {
            eta <- drop(z %*% b)
            (sum(pmax(eta, 0) + log1p(exp(-abs(eta))) - y*eta) + sum(b[-1]^2)/2)/length(y)
        }
        gr <- function(b) (drop(crossprod(z, plogis(z %*% b)-y)) + c(0, b[-1]))/length(y)
        fit <- optim(rep(0, ncol(z)), fn, gr, method = "BFGS",
                     control = list(maxit = 3000, reltol = 1e-10))
        stopifnot(fit$convergence == 0)
        return(list(nombre = nombre, escala = e, beta = fit$par,
                    convergencia = fit$convergence, objetivo = fit$value))
    }
    fit <- ranger::ranger(x = as.data.frame(x, check.names = FALSE),
        y = factor(y, levels = c(0,1)), probability = TRUE, num.trees = 150,
        max.depth = 8, min.node.size = 1, min.bucket = 5, mtry = floor(sqrt(ncol(x))),
        splitrule = "gini", replace = TRUE, sample.fraction = 1,
        seed = 20260924, num.threads = 1, oob.error = FALSE)
    list(nombre = nombre, modelo = fit)
}
predecir_probabilidad <- function(obj, x) {
    if (obj$nombre == "FRECUENCIA") return(rep(obj$media, nrow(x)))
    if (obj$nombre == "LOGISTICA_C1")
        return(drop(plogis(cbind(1, transformar(x, obj$escala)) %*% obj$beta)))
    drop(predict(obj$modelo, data = as.data.frame(x, check.names = FALSE),
                 num.threads = 1)$predictions[, "1"])
}
matriz <- function(fila, titulo) {
    plot.new(); plot.window(xlim = c(-.5,1.5), ylim = c(-.5,1.5), asp = 1)
    valores <- matrix(c(fila$VN, fila$FP, fila$FN, fila$VP), 2, byrow = TRUE)
    for (i in 1:2) for (j in 1:2) {
        xx <- j-1; yy <- 2-i
        rect(xx-.5, yy-.5, xx+.5, yy+.5, col = adjustcolor(if (i == j) AZUL else NARANJA, .25), border = NA)
        text(xx, yy, valores[i,j], cex = 1.3)
    }
    axis(1, 0:1, c("No spam", "Spam")); axis(2, 0:1, c("Spam", "No spam"))
    title(main = titulo, xlab = "Predicha", ylab = "Real")
}

for (nombre in modelos) {
    oof <- rep(NA_real_, length(y)); cobertura <- integer(length(y))
    sse_ajuste <- 0; n_ajuste <- 0
    for (k in 1:5) {
        valida <- entrenamiento$fold == k
        stopifnot(!length(intersect(entrenamiento$grupo[valida], entrenamiento$grupo[!valida])))
        ajuste <- ajustar_regresion(nombre, x[!valida, ], y[!valida])
        oof[valida] <- predecir_regresion(ajuste, x[valida, ]); cobertura[valida] <- cobertura[valida]+1L

        sse_ajuste <- sse_ajuste + sum((y[!valida]-pred_train)^2); n_ajuste <- n_ajuste+sum(!valida)
        por_fold[[length(por_fold)+1L]] <- data.frame(modelo = nombre, fold = k,
            n = sum(valida), as.data.frame(metricas(y[valida], oof[valida])))
    }
    stopifnot(all(cobertura == 1), all(is.finite(oof)))
    resumen[[nombre]] <- data.frame(modelo = nombre, orden = match(nombre, modelos)-1,
        as.data.frame(metricas(y,oof)), RMSE_ajuste = sqrt(sse_ajuste/n_ajuste))
    predicciones[[nombre]] <- data.frame(modelo = nombre, id = entrenamiento$id,
        grupo = entrenamiento$grupo, fold = entrenamiento$fold, real = y, predicha = oof)
}

mostrar(ranking); cat("Modelo elegido:",nombre_modelo(ganador),"\n")
figura("31_spam_modelos",function() {
    par(mfrow = c(1,2),mar = c(4,10,3,1))
    for (metrica in c("log_loss","Brier")) barplot(rev(ranking[[metrica]]),horiz = TRUE,
        names.arg = rev(nombre_modelo(ranking$modelo)),las = 1,col = c(AZUL,AZUL,VERDE),xlab = metrica)
})

# %% 09a. Clasificación: leer ROC y precision-recall ----
# Leer los ejes y el ordenamiento antes de mirar calibración.
curva <- curvas(y,p); medidas <- probabilisticas(y,p)
figura("32_spam_roc_pr",function() {
    par(mfrow = c(1,2))
    plot(curva$fpr,curva$recall,type = "l",col = AZUL,xlab = "Tasa de FP",ylab = "Recall",
         main = sprintf("ROC: AUC = %.3f",medidas$ROC_AUC)); abline(0,1,lty = 2,col = "gray")
    plot(curva$recall,curva$precision,type = "l",col = VERDE,xlab = "Recall",ylab = "Precision",
         ylim = c(0,1),main = sprintf("PR: AP = %.3f",medidas$AP)); abline(h = mean(y),col = NARANJA,lty = 2)
})
# %% 09b. Clasificación: comparar probabilidades y frecuencias ----
# Comparar probabilidad media y frecuencia observada.
bin <- cut(p,breaks = unique(quantile(p,seq(0,1,length.out = 6),type = 7)),include.lowest = TRUE)
cal <- do.call(rbind,lapply(split(seq_along(p),bin,drop = TRUE),function(i)
    data.frame(media_probabilidad = mean(p[i]),fraccion_spam = mean(y[i]),n = length(i))))
figura("33_spam_calibracion",function() {
    par(mfrow = c(1,2))
    plot(cal$media_probabilidad,cal$fraccion_spam,type = "b",pch = 19,col = AZUL,
        xlim = c(0,1),ylim = c(0,1),xlab = "Probabilidad media",ylab = "Fracción de spam",main = "Calibración OOF")
    abline(0,1,lty = 2,col = "gray")
    hist(p,breaks = seq(0,1,.1),col = AZUL,border = NA,xlab = "Probabilidad OOF",ylab = "Correos",main = "Distribución de probabilidades")
})
print(cal,row.names = FALSE)

# %% 10a. Clasificación: elegir el umbral por costo ----
# Leer el umbral elegido antes de examinar sus errores.
costos <- do.call(rbind,lapply(umbrales,function(t) as.data.frame(decision(y,p,t))))
eleccion <- costos[order(costos$costo,-costos$umbral), ][1, ]; umbral <- eleccion$umbral
cat("Umbral:",umbral,"Costo OOF:",eleccion$costo,"\n")
figura("34_spam_umbral",function() {
    plot(costos$umbral,1000*costos$costo,type = "b",pch = 19,col = AZUL,
        xlab = "Umbral",ylab = "Costo por 1000 correos",main = "FP=5; FN=1")
    abline(v = .5,lty = 2,col = "gray"); abline(v = umbral,lty = 2,col = NARANJA)
},8,4)
# %% 10b. Clasificación: leer el intercambio entre FP y FN ----
# Leer FP y FN; la prueba se evalúa recién en el tramo 11.
figura("35_spam_decisiones",function() {
    par(mfrow = c(1,2))
    for (t in c(.5,umbral)) { rr <- decision(y,p,t)
        matriz(rr,sprintf("OOF: t=%.2f; costo/1000=%.1f",t,1000*rr$costo)) }
})

# %% 11. Clasificación: reajustar y evaluar regla cerrada ----
# Evaluar una vez el modelo y umbral ya elegidos.
modelo_final <- ajustar_clasificador(ganador,x,y)
pt <- predecir_probabilidad(modelo_final,te[variables])
final <- data.frame(modelo = ganador,as.data.frame(c(probabilisticas(te$spam,pt),decision(te$spam,pt,umbral))))
referencia <- data.frame(modelo = "SIEMPRE_NO_SPAM",as.data.frame(decision(te$spam,rep(0,nrow(te)),.5)))

figura("36_spam_prueba",function() {

        col = c(NARANJA,AZUL),ylab = "Costo por 1000 correos",main = "Mismos costos declarados")
})
guardar("comparacion_cv",ranking); guardar("metricas_por_fold",por_fold); guardar("predicciones_oof",oof)

    grupo = te$grupo,real = te$spam,probabilidad = pt,predicha = as.integer(pt>=umbral)))
